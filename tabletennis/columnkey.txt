COLUMN KEY:
Ms = Matches
P1 BB% = Player 1 Bounce Back % (lose set 1, win set 2)
P1 BB# = Player 1 Bounce Back # (lose set 1 sample size)
P2 BB% = Player 2 Bounce Back % (lose set 1, win set 2)
P2 BB# = Player 2 Bounce Back # (lose set 1 sample size)
1ALL% = Tied 1-1 % (matches tied 1-1 after 2 sets)
P1 SR% = Player 1 Sweep Resistance % (lose sets 1 and 2, win set 3)
P1 SR# = Player 1 Sweep Resistance # (lose sets 1 and 2 sample size)
P2 SR% = Player 2 Sweep Resistance % (lose sets 1 and 2, win set 3)
P2 SR# = Player 2 Sweep Resistance # (lose sets 1 and 2 sample size)
NS% = Non-sweep % (matches that don't end with a 3-0 score)
P1 S = Player 1 Sweeps (matches won 3-0)
P2 S = Player 2 Sweeps (matches won 3-0)
3Set% = matches that lasted exactly 3 sets
4Set% = matches that lasted exactly 4 sets
5Set% = matches that lasted exactly 5 sets
ATS = Average Total Sets
SS = Set Spread
ATP = Average Total Points
PS = Point Spread
P1 W = Player 1 Wins
P2 W = Player 2 Wins
P1 W% = Player 1 Win %
